```typescript
export interface GroundTruthSegment {
  start: number;      // seconds
  end: number;        // seconds
  speaker: 'doctor' | 'patient';
  text_snippet: string;  // first few words for identification
}

export interface GroundTruth {
  audioFile: string;
  totalDurationSeconds: number;
  segments: GroundTruthSegment[];
}

export interface DeepgramUtterance {
  speaker: number;
  transcript: string;
  confidence: number;
  start: number;
  end: number;
  id: string;
  words: Array<{
    word: string;
    speaker: number;
    speaker_confidence: number;
    start: number;
    end: number;
    confidence: number;
  }>;
}

export interface ChunkResult {
  chunkIndex: number;
  chunkSizeSeconds: number;
  startTime: number;
  endTime: number;
  utterances: DeepgramUtterance[];
  rawResponse: unknown;
}

export interface SpikeResult {
  audioFile: string;
  chunkSizeLabel: string;    // "5s", "30s", "60s", "full"
  chunkSizeSeconds: number;  // 5, 30, 60, or total duration
  totalChunks: number;
  chunks: ChunkResult[];
  allUtterances: DeepgramUtterance[];
  timestamp: string;
}
```

**spike/validation-diarization/src/run-spike.ts:**
Main script that:

1. Loads `.env` with `dotenv/config` for DEEPGRAM_API_KEY
2. Accepts command-line args: `--audio <path>` (path to audio file in `audio/` dir)
3. Creates Deepgram client via `createClient(process.env.DEEPGRAM_API_KEY!)`
4. Defines chunk sizes: `[5, 30, 60]` plus full-file
5. For each chunk size:
   a. Uses `fluent-ffmpeg` with `@ffmpeg-installer/ffmpeg` (get path via `require('@ffmpeg-installer/ffmpeg').path` and set `ffmpeg.setFfmpegPath(...)`) to split audio into chunks. Use ffmpeg `-f segment -segment_time N` for splitting. Output chunks as WAV files (`-ar 16000 -ac 1`) to a temp directory.
   b. For each chunk, reads the WAV file into a Buffer
   c. Calls `deepgram.listen.prerecorded.transcribeFile(buffer, { model: 'nova-2', language: 'pt-BR', smart_format: true, punctuate: true, diarize: true, utterances: true })`
   d. Extracts utterances from `result.results?.utterances || []`
   e. Stores results in `ChunkResult` objects
6. For full-file test: reads entire audio, sends as single buffer (no splitting)
7. Saves each `SpikeResult` to `results/results-{chunkSize}.json`
8. Prints summary table to console: chunk size | total utterances | unique speakers found | utterances per speaker
9. Cleans up temp chunk files after processing

Important: Use `@ffmpeg-installer/ffmpeg` for the ffmpeg binary path (same pattern as the existing realtime-service). Do NOT rely on system ffmpeg. Import pattern:
```typescript
import ffmpegInstaller from '@ffmpeg-installer/ffmpeg';
import ffmpeg from 'fluent-ffmpeg';
ffmpeg.setFfmpegPath(ffmpegInstaller.path);
```

Error handling: If Deepgram returns an error for a chunk, log the error with chunk details and continue to next chunk (do not abort entire run). If DEEPGRAM_API_KEY is missing, exit with clear error message.

Console output during run: Print progress like "Processing 30s chunks: chunk 3/8..." so the user can see progress.

**spike/validation-diarization/src/analyze-results.ts:**
Script that:

1. Accepts args: `--results <dir>` (default: `./results`), `--ground-truth <path>` (default: `./ground-truth.json`)
2. Loads ground truth JSON (typed as `GroundTruth` from types.ts)
3. Loads all `results-*.json` files from results directory (typed as `SpikeResult[]`)
4. For each SpikeResult (each chunk size):
   a. Collects all utterances across chunks
   b. Determines speaker mapping: For each Deepgram speaker ID (0, 1), count how many utterance midpoints fall in ground truth "doctor" vs "patient" segments. The speaker ID with more doctor-segment overlap maps to "doctor". This handles the arbitrary speaker ID assignment per chunk.
   c. For the full-file result, mapping is straightforward (one consistent set of IDs). For chunked results, compute mapping PER CHUNK since speaker IDs reset per API call.
   d. Computes accuracy metrics:
      - `totalUtterances`: count of utterances matched to a ground truth segment
      - `correctSpeaker`: count where mapped speaker matches ground truth
      - `accuracy`: correctSpeaker / totalUtterances (as percentage)
      - `avgConfidence`: mean of all word-level `speaker_confidence` values
      - `highConfidenceAccuracy`: accuracy considering only utterances where avg word speaker_confidence > 0.7
      - `speakerDistribution`: count of utterances per speaker ID per chunk
      - `singleSpeakerChunks`: count of chunks where only 1 speaker was detected (indicates insufficient audio for diarization)
5. Outputs a comparison table to console:
   ```
   | Chunk Size | Utterances | Accuracy | Avg Confidence | High-Conf Accuracy | Single-Speaker Chunks |
   |------------|-----------|----------|----------------|--------------------|-----------------------|
   | 5s         | 42        | 61.2%    | 0.45           | 72.0%              | 8/12                  |
   | 30s        | 38        | 84.3%    | 0.78           | 91.0%              | 1/4                   |
   | 60s        | 35        | 87.5%    | 0.82           | 93.0%              | 0/2                   |
   | Full       | 34        | 89.2%    | 0.85           | 95.0%              | 0/1                   |
   ```
6. Outputs go/no-go recommendation based on threshold (default 85%):
   - If any chunk size >= 85% accuracy: "GO -- chunk size {N}s achieves {X}% accuracy"
   - If no chunk size >= 85% but full-file >= 85%: "CONDITIONAL GO -- full-file works ({X}%) but chunked accuracy is below threshold. Consider larger chunks or streaming approach."
   - If full-file < 85%: "NO-GO -- nova-2 diarization for pt-BR does not meet accuracy threshold even with full-file processing ({X}%)"
7. Saves full analysis to `results/analysis.json` with all metrics
8. The midpoint matching algorithm: for each utterance, compute midpoint = (start + end) / 2, find ground truth segment where midpoint falls within [segment.start, segment.end]. If no match found (gap between segments), skip utterance.
